import assert from 'node:assert/strict';
import {readFile, writeFile} from 'node:fs/promises';

import {chromium} from '/Users/oleksandr.mekhovov/.cache/codex-runtimes/codex-primary-runtime/dependencies/node/node_modules/playwright/index.mjs';
const secret=process.env.FPV_LIVE_TEST_PASSWORD;delete process.env.FPV_LIVE_TEST_PASSWORD;
assert.ok(secret);const [expected,output]=process.argv.slice(2);
const root='https://mekhovov.github.io/revealline/';
const report={format:'FPVPublicFreshPasswordAcceptance.v1',expectedRevision:expected,passed:false,packages:[],runnerSha256:createHash('sha256').update(await readFile(new URL(import.meta.url))).digest('hex'),limits:['Fresh isolated Chrome contexts on one desktop; normal public password form, no remembered unlock fixture.']};

  const context=await browser.newContext();try{
   const page=await context.newPage();const errors=[];page.on('pageerror',e=>errors.push(String(e.message).split(secret).join('[redacted]')));

   report.packages.push({id,initiallyLocked:true,passwordFormAccepted:true,ratesEditorOpened:true,pageErrors:errors});
  } finally {await context.close();}
 }
 const after=await (await fetch(root+'main-deployment.json?password-acceptance='+Date.now(),{signal:AbortSignal.timeout(30000)})).json();assert.equal(after.sourceRevision,expected);report.passed=true;
