import test from 'node:test';
import assert from 'node:assert/strict';
import { readFile } from 'node:fs/promises';
import { EXTERNAL_TMP_PARENT as parent, resolveBuildExternalTmp, prepareBuildExternalTmp, inspectBuildExternalTmp } from './final-build-external-tmp.mjs';
const root='/task/source',name='build-tmp-fixed-abcdef',target=parent+'/'+name,floor=200*1024**2;
const info=(dev,ino,extra={})=>({dev,ino,uid:502,mode:0o40700,isDirectory:()=>true,isSymbolicLink:()=>false,...extra});
function fixture(){const nodes=new Map([[root,info(1,1)],['/Volumes',info(1,2)],['/Volumes/alex',info(17,3)],[parent,info(17,37530202)]]),made=[],space=new Map([[root,floor+1],[parent,10*1024**3]]);const absent=()=>Object.assign(Error('missing'),{code:'ENOENT'});const fs={async realpath(p){if(!nodes.has(p))throw absent();return p;},async lstat(p){if(!nodes.has(p))throw absent();return nodes.get(p);},async statfs(p){if(!nodes.has(p))throw absent();return{bavail:space.get(p)??space.get(parent),bsize:1};},async mkdir(p,options){if(nodes.has(p))throw Object.assign(Error('exists'),{code:'EEXIST'});made.push({path:p,options});nodes.set(p,info(nodes.get(parent).dev,99,{mode:0o40000|options.mode}));}};return{nodes,made,space,fs,options:{fs,uid:502}};}
const prepare=f=>prepareBuildExternalTmp({root,directory:parent,runName:name,minimumFreeBytes:floor},f.options);
test('observes current device and binds a single fresh private external tmp child',async()=>{const f=fixture(),s=await prepare(f);assert.equal(s.device,17);assert.equal(s.parentInode,37530202);assert.equal(s.childInode,99);assert.equal(s.tmp,target);assert.deepEqual(f.made,[{path:target,options:{mode:0o700}}]);assert.equal((await inspectBuildExternalTmp(s,f.options)).availableBytes,10*1024**3);});
test('only the exact owned absolute parent and safe child name are accepted',async()=>{for(const directory of [undefined,'','.cache/tmp','/Volumes/alex/elsewhere',parent+'/../elsewhere',root])assert.throws(()=>resolveBuildExternalTmp(root,directory));for(const runName of ['../escape','build-tmp-../escape','build-tmp-','build-tmp-a/b']){const f=fixture();await assert.rejects(prepareBuildExternalTmp({root,directory:parent,runName,minimumFreeBytes:floor},f.options));assert.equal(f.made.length,0);}});
test('missing, symlinked, wrong-owned, replaced or local fallback parent fails before writes',async()=>{for(const mutate of [f=>f.nodes.delete(parent),f=>f.nodes.get('/Volumes/alex').isSymbolicLink=()=>true,f=>f.nodes.get(parent).uid=0,f=>f.nodes.get(parent).ino++,f=>f.nodes.get(parent).dev=1,f=>{f.options.uid=0;},f=>{const old=f.fs.realpath;f.fs.realpath=p=>p===parent?'/other':old(p);}]){const f=fixture();mutate(f);await assert.rejects(prepare(f));assert.equal(f.made.length,0);}});
test('both unchanged200MiB floors fail closed, including invalid capacity',async()=>{for(const where of [root,parent])for(const amount of [floor-1,NaN,-1]){const f=fixture();f.space.set(where,amount);await assert.rejects(prepare(f));assert.equal(f.made.length,0);}});
test('existing child is never reused or overwritten',async()=>{const f=fixture();f.nodes.set(target,info(17,123));await assert.rejects(prepare(f),/existing temporary child/);assert.equal(f.made.length,0);});
test('creation race or incorrect new child properties fail without deleting anything',async()=>{for(const mutate of [f=>{f.fs.mkdir=async()=>{throw Object.assign(Error('race'),{code:'EEXIST'});};},f=>{const old=f.fs.mkdir;f.fs.mkdir=async(...a)=>{await old(...a);f.nodes.get(target).mode=0o40755;};},f=>{const old=f.fs.mkdir;f.fs.mkdir=async(...a)=>{await old(...a);f.nodes.get(parent).dev=18;};}]){const f=fixture();mutate(f);await assert.rejects(prepare(f));assert(!('rm'in f.fs));}});
test('monitor rejects changed parent/child identity and symlink or owner replacements',async()=>{for(const mutate of [f=>f.nodes.get(parent).dev++,f=>f.nodes.get(parent).ino++,f=>f.nodes.get(target).ino++,f=>f.nodes.get(target).uid=0,f=>f.nodes.get(target).mode=0o40755,f=>f.nodes.get(target).isSymbolicLink=()=>true,f=>f.nodes.delete(target)]){const f=fixture(),state=await prepare(f);mutate(f);await assert.rejects(inspectBuildExternalTmp(state,f.options));}});
test('monitor exposes a low current capacity without silently lowering the threshold',async()=>{const f=fixture(),state=await prepare(f);f.space.set(parent,floor-1);assert.equal((await inspectBuildExternalTmp(state,f.options)).availableBytes,floor-1);});
test('observer retains canonical internal capture/jobs/budgets and disables late stop after close',async()=>{const old=await readFile(new URL('./run-final-build-current.mjs',import.meta.url),'utf8'),next=await readFile(new URL('./run-final-build-external-tmp.mjs',import.meta.url),'utf8');const jobs=s=>s.slice(s.indexOf('const jobs = ['),s.indexOf('for (const path of [reportPath'));assert.equal(jobs(next),jobs(old));assert(next.includes('const previewOutput = `.cache/shared-demos-emitted-${suffix}`;'));assert(next.includes('const minimumDiskBytes = 200 * 1024 * 1024;'));assert(next.includes('stageTimeoutMs: 20 * 60 * 1000'));assert(next.includes('TMPDIR: scratch.tmp, TMP: scratch.tmp, TEMP: scratch.tmp'));assert(next.includes('if (stopped || childClosed) return;'));assert(next.includes('if (monitoring || childClosed) return;'));assert(next.includes('childClosed = true; clearTimeout(timer); clearInterval(monitor);'));assert(next.includes('await inspectBuildReceipt(job.receipt)'));assert(next.includes('buildStageFailure(row, { requiresReceipt: Boolean(job.receipt) })'));});
