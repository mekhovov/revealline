name: Native demo catalog verification

on:
  pull_request:
    paths:
      - 'game/**'
      - 'scripts/capture-solo-native-candidates.mjs'
      - 'scripts/test-capture-solo-native-candidates.mjs'
      - 'scripts/qualify-solo-demos.mjs'
      - 'scripts/audit-solo-demo-scope.mjs'
      - 'scripts/build-demo-recordings.mjs'
      - 'scripts/qualify-survival-native.mjs'
      - '.github/workflows/solo-native-candidates.yml'

permissions:
  contents: read

jobs:
  native-catalog-verification:
    runs-on: ubuntu-24.04
    strategy:
      fail-fast: false
      matrix:
        node: ['20.19.6', '22.13.1']
    timeout-minutes: 32
    env:
      NATIVE_DEMO_NODE: ${{ matrix.node }}
      SOLO_CAPTURE_HEAD_SHA: ${{ github.event.pull_request.head.sha }}
      GIT_NO_LAZY_FETCH: '1'
      NODE_DISABLE_COMPILE_CACHE: '1'
    steps:
      - uses: actions/checkout@fbc6f3992d24b796d5a048ff273f7fcc4a7b6c09 # v5
        timeout-minutes: 3
        with:
          ref: ${{ github.event.pull_request.head.sha }}
          persist-credentials: false
          fetch-depth: 1
      - uses: actions/setup-node@a0853c24544627f65ddf259abe73b1d18a591444 # v5
        timeout-minutes: 2
        with:
          node-version: ${{ matrix.node }}
          package-manager-cache: false
      - name: Check strict collector guards
        timeout-minutes: 1
        run: node --test --test-concurrency=1 scripts/test-capture-solo-native-candidates.mjs
      - name: Verify every published Solo source without generating recordings
        id: verification
        timeout-minutes: 17
        shell: bash
        run: |
          set -euo pipefail
          test "$(git rev-parse HEAD)" = "$SOLO_CAPTURE_HEAD_SHA"
          test -z "$(git status --porcelain --untracked-files=no)"
          mkdir -p .cache
          mkdir .cache/solo-native-verification
          # The collector itself has a 15-minute total and 60-second worker budget.
          # Retain at most 4 MiB plus an overflow sentinel; any truncation fails.
          set +e
          timeout --signal=TERM --kill-after=10s 16m node scripts/capture-solo-native-candidates.mjs 2>&1 | head -c 4194305 > .cache/solo-native-verification/scan.log
          statuses=("${PIPESTATUS[@]}")
          set -e
          test "$(wc -c < .cache/solo-native-verification/scan.log)" -le 4194304
          test "${statuses[0]}" -eq 0
          test "${statuses[1]}" -eq 0
          node --input-type=module <<'NODE'
          import assert from 'node:assert/strict';
          import { readFile, stat } from 'node:fs/promises';
          assert.equal(process.version, `v${process.env.NATIVE_DEMO_NODE}`);
          assert.equal(process.platform, 'linux');
          assert.equal(process.arch, 'x64');
          const path = '.cache/solo-native-candidates/report.json';
          assert((await stat(path)).size <= 8 * 1024 ** 2);
          const report = JSON.parse(await readFile(path, 'utf8'));
          assert.equal(report.mode, 'scan-only');
          assert.equal(report.complete, true);
          assert.equal(report.scanComplete, true);
          assert.equal(report.sourceStable, true);
          assert.equal(report.existingCoveragePassed, true);
          assert.equal(report.sourceBefore.revision, process.env.SOLO_CAPTURE_HEAD_SHA);
          assert.deepEqual(report.sourceAfter, report.sourceBefore);
          assert.equal(report.sources.length, report.requiredSources);
          assert(report.requiredSources > 0 && report.requiredEntries.length > 0);
          assert(report.sources.every(source => source.strictUsable && !source.candidate));
          assert.deepEqual(report.missingEntries, []);
          assert.deepEqual(report.unusableSources, []);
          assert.deepEqual(report.unusableEntries, []);
          assert.deepEqual(report.artifacts, []);
          assert.equal(report.automaticallyAdmitted, false);
          NODE
      - name: Verify every authored Survival scenario without generating recordings
        id: survival
        if: ${{ !cancelled() && steps.verification.outcome != 'skipped' }}
        timeout-minutes: 6
        shell: bash
        run: |
          set -euo pipefail
          test "$(git rev-parse HEAD)" = "$SOLO_CAPTURE_HEAD_SHA"
          test -z "$(git status --porcelain --untracked-files=no)"
          mkdir -p .cache
          set +e
          timeout --signal=TERM --kill-after=10s 330s node scripts/qualify-survival-native.mjs --out=.cache/survival-native-qualification.json --timeout-ms=300000 2>&1 | head -c 262145 > .cache/survival-native-qualification.log
          statuses=("${PIPESTATUS[@]}")
          set -e
          test "$(wc -c < .cache/survival-native-qualification.log)" -le 262144
          test "${statuses[0]}" -eq 0
          test "${statuses[1]}" -eq 0
          test "$(git rev-parse HEAD)" = "$SOLO_CAPTURE_HEAD_SHA"
          test -z "$(git status --porcelain --untracked-files=no)"
          node --input-type=module <<'NODE'
          import assert from 'node:assert/strict';
          import { readFile, stat } from 'node:fs/promises';
          import { survivalDemoInventory } from './game/demo-modes/survival-catalog.mjs';
          const path = '.cache/survival-native-qualification.json';
          assert((await stat(path)).size <= 256 * 1024);
          const report = JSON.parse(await readFile(path, 'utf8'));
          assert.equal(report.format, 'SurvivalNativeQualificationV1');
          assert.equal(report.complete, true);
          assert.equal(report.head, process.env.SOLO_CAPTURE_HEAD_SHA);
          assert.equal(report.environment.node, `v${process.env.NATIVE_DEMO_NODE}`);
          assert.equal(report.environment.platform, 'linux');
          assert.equal(report.environment.arch, 'x64');
          const ids = survivalDemoInventory().map(entry => entry.id);
          assert(ids.length > 0);
          assert.deepEqual(report.checks.map(entry => entry.id), ids);
          assert.deepEqual(report.counts, { required: ids.length, checked: ids.length, verified: ids.length });
          assert(report.checks.every(entry => entry.verified && ['won', 'lost'].includes(entry.outcome) && entry.progress.kills > 0 && entry.progress.xpCollected > 0 && entry.progress.upgrades > 0));
          NODE
      - name: Retain strict verification and bounded partial diagnostics
        if: ${{ always() && steps.verification.outcome != 'skipped' }}
        timeout-minutes: 2
        uses: actions/upload-artifact@b7c566a772e6b6bfb58ed0dc250532a479d7789f # v6
        with:
          name: native-demo-verification-node-${{ matrix.node }}-${{ github.event.pull_request.head.sha }}-${{ github.run_attempt }}
          path: |
            .cache/solo-native-candidates/report.json
            .cache/solo-native-verification/scan.log
            .cache/survival-native-qualification.json
            .cache/survival-native-qualification.log
          include-hidden-files: true
          if-no-files-found: error
          retention-days: 7
