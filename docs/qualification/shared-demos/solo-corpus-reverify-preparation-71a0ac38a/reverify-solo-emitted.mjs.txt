const sha = body => createHash('sha256').update(body).digest('hex');
const catalogNames = ['demo-data/catalog.json', 'demo-data/catalog.journey-current.json', 'demo-data/catalog.journey-archive.json'];
const engines = ['chromium', 'firefox', 'webkit'];
const qualifier = 'scripts/qualify-solo-browser.mjs';
const helperNames = ['.cache/reverify-solo-emitted.mjs', '.cache/reverify-solo-emitted.py', '.cache/demo-probe-evidence.mjs', '.cache/run-endurance-emitted-final.py'];

export function validateResult(result, expected, { engine, reviewed, head, base }) {
  assert.equal(result.complete, true);
  assert.equal(result.engine, engine);
  assert.equal(result.sourceRevision, head);
  assert.equal(result.mainRevision, base);
  assert.ok(result.started && result.finished && result.browserVersion);
  assert.equal(result.error, undefined);
  assert.deepEqual(result.failures, []);
  assert.deepEqual(result.variants, []);
  assert.deepEqual(result.performances, []);
  assert.equal(result.required, expected.length);
  const rows = reviewed ? result.runtime : result.verified;
  assert.ok(Array.isArray(rows));
  assert.deepEqual(rows.map(row => row.id).sort(), expected.map(row => row.id).sort(), 'Exact catalog IDs, once each.');
  for (const row of rows) {
    const clip = expected.find(clip => clip.id === row.id);
    assert.ok(['won', 'lost'].includes(row.outcome), `${row.id}: terminal result required`);
    assert.ok(Number.isInteger(row.captures) && row.captures > 0, `${row.id}: captures required`);
    assert.ok(Number.isInteger(row.capturedCells) && row.capturedCells > 0, `${row.id}: claimed progress required`);
    if (reviewed) {
      assert.equal(row.ok, true);
      assert.ok(row.themes > 0);
      assert.ok(clip.candidates.some(candidate => candidate.ticks === row.tick && candidate.outcome === row.outcome), `${row.id}: native playback must reach a published candidate's terminal tick/outcome`);
    } else {
      assert.equal(row.variant, false, 'No recordings may be regenerated.');
      assert.equal(row.independent, false);
      const selected = clip.selected[engine];
      assert.equal(row.ticks, selected.ticks);
      assert.equal(row.outcome, selected.outcome);
      assert.equal(row.checkpoint, selected.checkpoint);
    }
  }
  return { clips: rows.length, entries: new Set(expected.map(row => row.levelKey)).size, terminalAndMeaningfulProgress: true };
}

export function parseArgs(values) {
  const result = {};
  const allowed = new Set(['run', 'head', 'base', 'fingerprint', 'files', 'bytes', 'emitted-preview', 'report', 'runtime', 'chromium', 'firefox', 'webkit', 'tmpdir', 'port']);
  for (const value of values) {
    const match = /^--([^=]+)=(.+)$/.exec(value);
    assert.ok(match && allowed.has(match[1]) && !Object.hasOwn(result, match[1]), `Unsupported or repeated argument: ${value}`);
    result[match[1]] = match[2];
  }
  assert.equal(result.run, 'approved', 'Await explicit browser/CPU release.');
  for (const name of ['head', 'base']) assert.match(result[name] ?? '', /^[a-f0-9]{40}$/);
  assert.match(result.fingerprint ?? '', /^[a-f0-9]{64}$/);
  for (const name of ['files', 'bytes']) assert.ok(Number.isSafeInteger(Number(result[name])) && Number(result[name]) > 0);
  for (const name of ['emitted-preview', 'report', 'runtime', 'tmpdir', ...engines]) assert.ok(result[name], `Required: ${name}`);
  const port = Number(result.port ?? 18777);
  assert.ok(Number.isInteger(port) && port > 1024 && port < 65536);
  return { ...result, port };
}

async function main() {
  const args = parseArgs(process.argv.slice(2));
  assert.equal(process.env.SOLO_REVERIFY_SUPERVISED, '1', 'Run through the bounded Python supervisor.');
  const root = process.cwd();
  const tmpdir = await realpath(args.tmpdir);
  assert.equal(await realpath(process.env.TMPDIR ?? ''), tmpdir, 'Use the explicitly owned local TMPDIR.');
  assert.ok((await lstat(tmpdir)).isDirectory());
  const reportPath = path.resolve(args.report);
  const reportBase = reportPath.replace(/\.json$/, '');
  assert.notEqual(reportPath, reportBase, 'Report must be JSON.');
  const jobs = engines.flatMap(engine => [false, true].map(reviewed => ({ engine, reviewed, id: `${engine}-${reviewed ? 'reviewed-native' : 'qualified-strict'}`, reportPath: `${reportBase}.${engine}.${reviewed ? 'reviewed-native' : 'qualified-strict'}.json` })));
  for (const name of [reportPath, ...jobs.map(job => job.reportPath)]) {
    try { await lstat(name); throw new Error(`Refusing existing evidence: ${name}`); }
    catch (error) { if (error.code !== 'ENOENT') throw error; }
  }
  const git = (...argv) => execFileSync('git', argv, { encoding: 'utf8', env: { ...process.env, GIT_NO_LAZY_FETCH: '1' } }).trim();
  const identity = { sha256: args.fingerprint, files: Number(args.files), bytes: Number(args.bytes) };
  const declarationPath = 'docs/qualification/shared-demos/source-freeze-latest.json';
  async function pins() {
    const files = [...execFileSync('rg', ['--files', 'scripts'], { encoding: 'utf8' }).trim().split('\n'), ...helperNames, declarationPath].sort();
    return Promise.all(files.map(async name => ({ path: name, sha256: sha(await readFile(name)) })));
  }
  const report = {
    format: 'SoloFullCorpusEmittedReverificationV1', complete: false, started: new Date().toISOString(),
    head: args.head, base: args.base, runtime: identity, jobs: [],
    scope: 'Solo only; exact current catalog inventory. Three engines sequentially. Each uses published strict replay verification/playback for qualified clips and the unchanged native reviewed-recording compatibility path. No recording generation or regeneration.',
    limitations: ['Uses the unchanged synthetic qualification document, not the ordinary UI. Does not establish rendering, controller, audio, physical-device or offline UX behavior.', 'Historical native Journey source admission/presentation remains separate evidence. This run rechecks full simulation playback, not a new native authored-route presentation run.', 'Original qualifier records verification failures but has no pageerror observer. No zero-browser-console-error claim is made.', 'For reviewed native recordings the existing qualifier reports the terminal tick/outcome, not selected payload identity. All allowed candidate bytes/identities are bound; a matching candidate is required without claiming which candidate was selected.'],
    bounds: { childSeconds: 600, supervisorSeconds: 2700, childCount: jobs.length, noRetries: true },
  };
  let preview, beforePins, emittedBefore, timedOutChild = false;
  try {
    assert.equal(git('rev-parse', 'HEAD'), args.head);
    git('merge-base', '--is-ancestor', args.base, args.head);
    const declaration = JSON.parse(await readFile(declarationPath, 'utf8'));
    assert.equal(declaration.integrationBase, args.base);
    assert.equal(declaration.sourceFingerprint, args.fingerprint);
    assert.equal(declaration.files, identity.files);
    assert.equal(declaration.bytes, identity.bytes);
    assert.deepEqual(await fingerprintDemoSources(root), identity);
    report.beforePins = beforePins = await pins();
    report.emittedBefore = emittedBefore = await emittedEvidenceSnapshot(root, args['emitted-preview']);
    assert.equal(emittedBefore.identity.sourceRevision, args.head);
    assert.deepEqual(emittedBefore.identity.sourceFingerprint, identity);
    const emitted = await openEmittedPreview(args['emitted-preview'], { root, verifySource: true });
    async function sameBytes(name) {
      const source = await readFile(name), published = await emitted.read(name);
      assert.deepEqual(published, source, `${name}: local qualifier input must equal emitted bytes`);
      return { body: source, pin: { path: name, bytes: source.length, sha256: sha(source) } };
    }
    const clips = [];
    report.catalogs = [];
    for (const name of catalogNames) {
      const file = await sameBytes(`game/${name}`);
      report.catalogs.push(file.pin);
      clips.push(...validateDemoCatalog(file.body.toString()).clips);
    }
    assert.equal(new Set(clips.map(clip => clip.id)).size, clips.length);
    const inventory = [];
    for (const clip of clips) {
      const candidates = [{ ...clip }, ...(clip.replayVariants ?? []).map(replayURL => ({ ...clip, replayURL })), ...(clip.runtimePerformances ?? []).map(performance => ({ ...clip, ...performance }))];
      const row = { id: clip.id, levelKey: `${clip.campaignKey}/${clip.levelId}`, reviewed: !clip.id.startsWith('solo-qualified-'), recordingIdentity: clip.recordingIdentity, candidates: [], selected: {} };
      for (const candidate of candidates) {
        const name = `game/${candidate.replayURL.replace(/^\.\//, '')}`;
        const { body, pin } = await sameBytes(name);
        const replay = snapshotReplay(body.toString());
        assert.equal(replay.level.id, clip.levelId);
        assert.equal(demoIdentity(replay.level, replay.options.classRecipes), clip.recordingIdentity);
        assert.equal(demoInputTraceIdentity(replay), candidate.inputTraceIdentity);
        assert.equal(replay.ticks / 120, candidate.durationSeconds);
        assert.equal(replay.options.turnPolicy, candidate.turnPolicy);
        assert.ok(['won', 'lost'].includes(replay.summary.status));
        const descriptor = { ...pin, replayURL: candidate.replayURL, inputTraceIdentity: candidate.inputTraceIdentity, ticks: replay.ticks, outcome: replay.summary.status, checkpoint: replay.checkpoint.hash };
        const diagnosticPath = name.replace(/\.replay\.json$/, '.diagnostics.json');
        if (emitted.has(diagnosticPath)) descriptor.diagnostic = (await sameBytes(diagnosticPath)).pin;
        row.candidates.push(descriptor);
      }
      for (const engine of engines) {
        const url = clip.runtimePerformances?.find(candidate => candidate.engine === engine)?.replayURL ?? clip.replayURL;
        row.selected[engine] = row.candidates.find(candidate => candidate.replayURL === url);
        assert.ok(row.selected[engine]);
      }
      inventory.push(row);
    }
    report.inventory = inventory;
    report.required = { entries: new Set(inventory.map(row => row.levelKey)).size, clips: inventory.length, qualified: inventory.filter(row => !row.reviewed).length, reviewed: inventory.filter(row => row.reviewed).length };
    assert.ok(report.required.qualified > 0 && report.required.reviewed > 0);
    preview = await startProductionPreview({ root, port: args.port, fixtureAccess: false, cache: 'revalidate', emittedPreview: args['emitted-preview'], verifyEmittedSource: true });
    report.preview = { url: preview.url, emittedPreview: preview.emittedPreview, fixtureAccess: preview.fixtureAccess };
    assert.equal(preview.emittedPreview.sourceVerified, true);
    assert.equal(preview.emittedPreview.sourceRevision, args.head);
    assert.equal(preview.emittedPreview.manifestSHA256, emittedBefore.manifest.sha256);
    await writeQualificationReport(reportPath, report);
    for (const job of jobs) {
      const argv = [qualifier, `--runtime=${args.runtime}`, `--browser=${job.engine}`, `--executable=${args[job.engine]}`, `--url=${preview.url.replace(/\/$/, '')}`, `--report=${job.reportPath}`, `--catalog=${catalogNames.join(',')}`, `--source-revision=${args.head}`, `--main-revision=${args.base}`, ...(job.reviewed ? ['--reviewed=true', '--runtime-sources=true'] : ['--published=true'])];
      const record = { ...job, argv: [process.execPath, ...argv], started: new Date().toISOString(), complete: false };
      report.jobs.push(record);
      await writeQualificationReport(reportPath, report);
      const child = spawn(process.execPath, argv, { stdio: 'inherit', env: { ...process.env, NODE_DISABLE_COMPILE_CACHE: '1' } });
      record.pid = child.pid;
      let timer;
      try {
        record.exit = await new Promise((resolve, reject) => {
          timer = setTimeout(() => { timedOutChild = true; reject(new Error(`${job.id} exceeded 600 seconds; stop and let the identity-aware supervisor clean owned descendants.`)); }, 600000);
          child.once('error', reject);
          child.once('close', (code, signal) => resolve({ code, signal }));
        });
      } finally { clearTimeout(timer); }
      const body = await readFile(job.reportPath);
      record.reportSHA256 = sha(body);
      record.exitObserved = new Date().toISOString();
      assert.deepEqual(record.exit, { code: 0, signal: null });
      record.coverage = validateResult(JSON.parse(body), inventory.filter(row => row.reviewed === job.reviewed), { ...job, head: args.head, base: args.base });
      record.complete = true;
      record.finished = new Date().toISOString();
      await writeQualificationReport(reportPath, report);
      console.log(JSON.stringify({ job: job.id, complete: true, ...record.coverage }));
    }
    report.complete = report.jobs.length === jobs.length && report.jobs.every(job => job.complete);
  } catch (error) { report.failure = String(error.stack ?? error).slice(0, 12000); report.complete = false; }
  finally {
    await closeOwnedPreview(preview, report);
    try {
      assert.equal(git('rev-parse', 'HEAD'), args.head);
      assert.deepEqual(await fingerprintDemoSources(root), identity);
      report.afterPins = await pins();
      assert.deepEqual(report.afterPins, beforePins);
      report.emittedAfter = await emittedEvidenceSnapshot(root, args['emitted-preview']);
      assert.deepEqual(report.emittedAfter, emittedBefore);
      report.bindingsStable = true;
    } catch (error) { report.complete = false; report.bindingFailure = String(error.stack ?? error).slice(0, 6000); }
    report.finished = new Date().toISOString();
    await writeQualificationReport(reportPath, report);
    console.log(JSON.stringify({ report: reportPath, complete: report.complete, required: report.required, failure: report.failure }));
  }
  if (timedOutChild) process.exit(1); // The outer owner cleans exact observed descendants; no later job starts.
  process.exitCode = report.complete ? 0 : 1;
}

if (process.argv[1] && import.meta.url === pathToFileURL(path.resolve(process.argv[1])).href) await main();
