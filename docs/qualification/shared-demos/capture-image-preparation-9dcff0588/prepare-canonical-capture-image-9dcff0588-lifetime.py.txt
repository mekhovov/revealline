def sha(body): return hashlib.sha256(body).hexdigest()
def free(path):
    value = os.statvfs(path)
    return value.f_bavail * value.f_frsize
def identity(path):
    value = path.lstat()
    assert stat.S_ISDIR(value.st_mode) and not path.is_symlink() and path.resolve(strict=True) == path
    return {'device': value.st_dev, 'inode': value.st_ino, 'uid': value.st_uid, 'mode': stat.S_IMODE(value.st_mode)}
def file_pin(path):
    body = path.read_bytes()
    return {'bytes': len(body), 'sha256': sha(body)}
def git_head():
    return subprocess.check_output(['git', '-c', 'maintenance.auto=false', '-c', 'gc.auto=0', 'rev-parse', 'HEAD'], env=env, text=True, timeout=10).strip()

def stable():
    assert identity(external) == parent_identity and identity(external.parent) == mount_parent_identity
    assert identity(capture) == source_identity
    assert free(root) >= 200 * 1024**2 and free(external) >= 512 * 1024**2

    for item in held_image_owners:
        item['owner'].capture()
        item['owner'].check_limits()
        item['log'].check_limits()

def interrupt(signum, frame):
    if active['registration']:
        active['pending'] = signum
        return
    report['interruptedSignal'] = signal.Signals(signum).name
    raise InterruptedError(report['interruptedSignal'])
signal.signal(signal.SIGINT, interrupt)
signal.signal(signal.SIGTERM, interrupt)

def command(label, argv, seconds, retain_until_detach=False):
    stage = {'label': label, 'command': argv, 'timeoutSeconds': seconds, 'complete': False}
    report['phases'].append(stage)
    path = report_path.with_suffix('.' + label + '.log.txt')
    assert not path.exists()
    log = BoundedChildLog(path, byte_limit=1024*1024, tail_limit=4096)
    child = owned = remaining = None
    begin = time.monotonic()
    try:
        active['registration'] = True
        try:
            child = subprocess.Popen(argv, cwd=root, env=env, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, start_new_session=True)
            owned = OwnedProcessTree(child)
            owned.capture()
            log.start(child.stdout)
        finally:
            active['registration'] = False
            pending = active['pending']; active['pending'] = None
            if pending is not None: interrupt(pending, None)
        while child.poll() is None:
            owned.capture(); owned.check_limits(); log.check_limits()
            if time.monotonic() - begin >= seconds: raise TimeoutError(label + ' exceeded its command budget.')
            time.sleep(.2)
        remaining = owned.wait_clear(.2 if retain_until_detach else 3)
        if remaining and not retain_until_detach:
            stage['cleanupNeeded'] = True
            remaining = owned.cleanup()
    except BaseException as error:
        stage['failure'] = repr(error)
    finally:
        previous_handlers = {s: signal.getsignal(s) for s in (signal.SIGINT, signal.SIGTERM)}
        def finish_command_interrupt(signum, frame):
            stage.setdefault('interruptedSignal', signal.Signals(signum).name)
            report.setdefault('interruptedSignal', signal.Signals(signum).name)
            report['complete'] = False
        for signum in previous_handlers: signal.signal(signum, finish_command_interrupt)
        if owned is not None:
            if retain_until_detach:
                # hdiutil attach's daemon owns the mounted image until explicit detach.
                # Transfer observation to the image lifetime; never kill that daemon here.
                if child.poll() is None:
                    current = owned.capture()
                    spawned = next((row for row in current if row['pid'] == child.pid), None)
                    if spawned: owned.signal(spawned, signal.SIGTERM)
                    owned.wait_clear(3)
                    if child.poll() is None:
                        current = owned.capture()
                        spawned = next((row for row in current if row['pid'] == child.pid), None)
                        if spawned: owned.signal(spawned, signal.SIGKILL)
                        owned.wait_clear(1)
                remaining = owned.capture()
                held_image_owners.append({'owner': owned, 'log': log, 'stage': stage, 'path': path})
                stage['ownerRetainedUntilExplicitDetach'] = True
            elif remaining is None:
                try: remaining = owned.cleanup()
                except BaseException as error: stage['cleanupFailure'] = repr(error)
            stage['ownedProcesses'] = owned.evidence(remaining)
        if retain_until_detach:
            # The daemon may retain its output pipe; final log completion is checked after detach.
            if not log.output.closed: log.output.flush()
        else:
            log.finish()
        stage['log'] = {'path': str(path.relative_to(root)), **log.evidence()}
        stage['exitCode'] = child.poll() if child else None
        stage['elapsedSeconds'] = time.monotonic() - begin
        owner_admitted = stage.get('ownedProcesses', {}).get('verified') is True
        if retain_until_detach:
            owner_admitted = stage.get('ownedProcesses', {}).get('rootObserved') is True and child.poll() is not None
        stage['complete'] = stage['exitCode'] == 0 and owner_admitted and not stage['ownedProcesses']['observationLimits']['exceeded'] and (stage['log']['complete'] or retain_until_detach) and not stage['log']['error'] and not stage['log']['limitExceeded'] and not any(stage.get(k) for k in ('failure', 'cleanupFailure', 'cleanupNeeded', 'interruptedSignal'))
        for signum, handler in previous_handlers.items(): signal.signal(signum, handler)
    assert stage['complete'], stage
    return path.read_bytes()

def images():
    result = subprocess.run(['hdiutil', 'info', '-plist'], env=env, stdin=subprocess.DEVNULL, capture_output=True, timeout=20)
    assert result.returncode == 0 and len(result.stdout) <= 1024**2
    return plistlib.loads(result.stdout).get('images', [])

def matching_image():
    found = [item for item in images() if Path(item.get('image-path', '')) == image]
    assert len(found) <= 1
    return found[0] if found else None

def retire_image_owners(detached):
    assert detached, 'Never retire a registered image owner before explicit detach.'
    report['imageOwnersAfterDetach'] = []
    for item in held_image_owners:
        remaining = item['owner'].wait_clear(3)
        if remaining:
            item['stage']['postDetachCleanupNeeded'] = True
            remaining = item['owner'].cleanup()
        item['log'].finish()
        item['stage']['finalLog'] = {'path': str(item['path'].relative_to(root)), **item['log'].evidence()}
        evidence = item['owner'].evidence(remaining)
        item['stage']['ownerAfterDetach'] = evidence
        report['imageOwnersAfterDetach'].append(evidence)
        assert evidence['verified'] and not evidence['observationLimits']['exceeded']
        assert item['stage']['finalLog']['complete'] and not item['stage']['finalLog']['error'] and not item['stage']['finalLog']['limitExceeded']

def inventory(directory):
    rows = []
    for current, dirs, files in os.walk(directory, followlinks=False):
        current = Path(current)
        for name in sorted(dirs):
            p = current / name
            value = p.lstat()
            assert stat.S_ISDIR(value.st_mode) and not p.is_symlink()
            rows.append({'path': str(p.relative_to(directory)), 'kind': 'directory', 'mode': stat.S_IMODE(value.st_mode)})
        for name in sorted(files):
            stable()
            p = current / name; value = p.lstat()
            assert stat.S_ISREG(value.st_mode) and not p.is_symlink() and value.st_uid == os.getuid()
            body = p.read_bytes(); after = p.lstat()
            assert (value.st_dev, value.st_ino, value.st_size, value.st_mtime_ns) == (after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns)
            rows.append({'path': str(p.relative_to(directory)), 'kind': 'file', 'mode': stat.S_IMODE(value.st_mode), 'bytes': len(body), 'sha256': sha(body)})
    return sorted(rows, key=lambda r: r['path'])
