async function fingerprint(directories, filter) {
  const paths = execFileSync('rg', ['--files', ...directories], { encoding: 'utf8', maxBuffer: 8 * 1024 * 1024 }).trim().split('\n').filter(filter).sort();
  const hash = createHash('sha256');
  const digests = new Map();
  let bytes = 0;
  for (const path of paths) {
    const value = await readFile(path), digest = sha(value);
    bytes += value.length;
    digests.set(path, digest);
    hash.update(path + '\0').update(Buffer.from(digest, 'hex'));
  }
  return { sha256: hash.digest('hex'), files: paths.length, bytes, digests };
}

async function closePreview(preview, record) {
  if (!preview) return;
  const started = Date.now();
  record.provider.fixtureResponses = preview.fixtureResponses;
  try {
    preview.server.closeAllConnections();
    await new Promise((resolve, reject) => {
      const timer = setTimeout(() => reject(new Error('Owned emitted preview cleanup exceeded2000ms.')), 2000);
      preview.server.close(error => {
        clearTimeout(timer);
        if (error) reject(error); else resolve();
      });
    });
    record.provider.closeCompleted = true;
  } catch (error) {
    record.provider.closeCompleted = false;
    record.provider.closeFailure = String(error.stack ?? error);
    throw error;
  } finally {
    record.provider.closeElapsedMs = Date.now() - started;
    record.provider.closedAt = new Date().toISOString();
  }
}
