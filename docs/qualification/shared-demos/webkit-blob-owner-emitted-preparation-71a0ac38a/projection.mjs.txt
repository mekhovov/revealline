/** Capture one document and error WeakMap; late calls cannot resolve a new document's sink. */
export function installFinalOwnerObserver() {
  const documentIdentity = String(performance.timeOrigin) + '-' + Math.random();
  const doc = document, sendNative = console.debug.bind(console), errors = new WeakMap();
  let errorId = 0, hostId = 0, sequence = 0, observerFailures = 0, overflow = 0;
  const rows = [];
  const safe = (work) => { try { return work(); } catch { observerFailures++; return undefined; } };
  const errorIdentity = (error) => {
    const object = error !== null && (typeof error === 'object' || typeof error === 'function');
    if (object && !errors.has(error)) errors.set(error, ++errorId);
    return { errorId: object ? errors.get(error) : null, name: String(error?.name ?? typeof error).slice(0, 160), message: String(error?.message ?? error).slice(0, 1400), stack: String(error?.stack ?? '').slice(0, 3000) };
  };
  const emit = (kind, data = {}) => safe(() => {
    if (rows.length >= 1800) {
      overflow++;
      if (overflow === 1) sendNative('DEMO_FINAL_OWNER:' + JSON.stringify({ documentIdentity, kind: 'observer-overflow', at: Date.now(), overflow }));
      return;
    }
    const row = { documentIdentity, sequence: ++sequence, kind, at: Date.now(), monotonic: performance.now(), timeOrigin: performance.timeOrigin, readyState: doc.readyState, ...data };
    rows.push(row);
    sendNative('DEMO_FINAL_OWNER:' + JSON.stringify(row));
  });
  const observer = Object.freeze({
    snapshot: () => ({ documentIdentity, sequence, observerFailures, overflow, rows: [...rows] }),
    appCatch: (phase, error) => safe(() => emit('app-catch', { phase, ...errorIdentity(error) })),
    host: (profile) => safe(() => {
      const host = ++hostId;
      let loadId = 0, requestId = 0, operationId = 0;
      const loads = new WeakMap();
      const record = (signal) => {
        if (!signal || (typeof signal !== 'object' && typeof signal !== 'function')) return null;
        if (!loads.has(signal)) loads.set(signal, { load: ++loadId, requests: [] });
        return loads.get(signal);
      };
      emit('host-create', { host, profile });
      return Object.freeze({
        operation: (signal, detail) => safe(() => {
          if (operationId >= 500) { observerFailures++; return undefined; }
          const { kind: operationKind, ...metadata } = detail;
          const operation = { ...metadata, operationKind, operation: ++operationId, load: record(signal)?.load ?? null };
          emit('operation-created', { host, ...operation, aborted: !!signal?.aborted });
          return operation;
        }),
        submitted: (operation, signal) => safe(() => emit('operation-submitted', { host, ...operation, aborted: !!signal?.aborted })),
        load: (signal) => safe(() => emit('load-start', { host, load: record(signal)?.load ?? null, aborted: !!signal?.aborted })),
        request: (signal, url) => safe(() => {
          const load = record(signal);
          if ((load?.requests.length ?? 0) >= 1000) { observerFailures++; return undefined; }
          const request = { request: ++requestId, load: load?.load ?? null, url, responseDelivered: false };
          load?.requests.push(request);
          emit('request-start', { host, ...request, aborted: !!signal?.aborted });
          return request;
        }),
        response: (request, status) => safe(() => {
          if (!request) return;
          request.responseDelivered = true;
          emit('response-delivered', { host, ...request, status });
        }),
        caught: (phase, signal, error, detail = {}) => safe(() => {
          const load = record(signal);
          emit('host-catch', { host, load: load?.load ?? null, phase, aborted: !!signal?.aborted, pendingRequests: (load?.requests ?? []).filter(r => !r.responseDelivered).map(r => ({ ...r })), ...errorIdentity(error), ...detail });
        }),
        settled: (signal, detail) => safe(() => emit('load-finally', { host, load: record(signal)?.load ?? null, aborted: !!signal?.aborted, ...detail })),
        close: (phase, detail) => safe(() => emit('host-close', { host, phase, ...detail })),
      });
    }),
  });
  Object.defineProperty(globalThis, '__demoFinalOwnerObserver', { value: observer, configurable: false, writable: false });
  for (const name of ['beforeunload', 'pagehide', 'pageshow'])
    addEventListener(name, (event) => emit('lifecycle', { event: name, persisted: event.persisted ?? null }));
  // These listeners observe existing events only: no preventDefault, new catch,
  // native codec wrapper or promise substitution. WebKit may emit neither event.
  addEventListener('error', event => safe(() => emit('dom-error', {
    ...errorIdentity(event.error), eventMessage: String(event.message ?? '').slice(0, 1400),
    filename: String(event.filename ?? '').slice(0, 1000), lineno: event.lineno, colno: event.colno,
    hasErrorObject: event.error !== null && (typeof event.error === 'object' || typeof event.error === 'function'),
  })));
  addEventListener('unhandledrejection', event => safe(() => emit('dom-unhandledrejection', { ...errorIdentity(event.reason) })));
  emit('document-start', { url: location.href, top: window === window.top, frameElement: window.frameElement?.id ?? null });
}


/** Completeness of diagnostic collection is separate from the unchanged strict pass. */
export function observationCollectionComplete(report, result) {
  const exit = report.childExit;
  const streams = report.streamClosure;
  const normalPass = exit?.code === 0 && result.complete === true &&
    result.functionalComplete === true && result.strictDiagnosticGate === true;
  const collectedStrictFailure = exit?.code === 1 && result.complete === false &&
    result.functionalComplete === true && result.strictDiagnosticGate === false;
  return report.watchdogExpired !== true && !report.terminationError &&
    !exit?.error && !exit?.signal && streams?.complete === true &&
    streams.code === exit?.code && !streams.signal &&
    (normalPass || collectedStrictFailure) &&
    report.bindingsUnchanged === true &&
    result.browserCloseCompleted === true && result.previewClosed === true &&
    result.observationPrefixComplete === true &&
    report.protocolOverflow === 0 && report.parseFailures === 0;
}
