async function runtimeFingerprint() {
  const paths = execFileSync('rg', ['--files', 'game', 'optional-practice', 'publishing'], {
    encoding: 'utf8',
    maxBuffer: 8 * 1024 * 1024,
  }).split('\n').filter(path =>
    /\.(mjs|js|json|css|html)$/.test(path) &&
    !path.includes('/test/') &&
    !path.endsWith('.test.mjs') &&
    !path.endsWith('/coverage.json')
  ).sort();
  const hash = createHash('sha256');
  let bytes = 0;
  for (const path of paths) {
    const value = await readFile(path);
    bytes += value.length;
    hash.update(path + '\0').update(createHash('sha256').update(value).digest());
  }
  return { sha256: hash.digest('hex'), files: paths.length, bytes };
}

}

async function ordinaryEntry(entry, { openMenu = false } = {}) {
  const menu = {
    solo: '#shell-menu',
    snake: '#snake-action-menu',
    sim: '#worlds-shell-action-menu',
    survival: '#overflight-action-menu',
  }[entry.mode];
  await page.waitForFunction(
    ({ button, menu }) => {
      if (globalThis.RevealLineBoot && document.documentElement.dataset.bootState !== 'ready')
        return false;
      const visible = (node) =>
        !!node?.getClientRects().length && getComputedStyle(node).visibility !== 'hidden';
      return (
        visible(document.querySelector(button)) ||
        (visible(document.querySelector(menu)) && !document.querySelector('dialog[open]'))
      );
    },
    { button: entry.button, menu },
    { timeout: 30000 },
  );
  assert.equal(await page.evaluate(() => document.documentElement.dataset.accessState), 'unlocked',
    'Exact fixture credential must unlock without editing a form in the discard target.');
  if (openMenu && !(await page.locator(entry.button).isVisible())) {
    try {
      await page.locator(menu).click({ timeout: 3000 });
    } catch (error) {
      if (!(await page.locator(entry.button).isVisible())) throw error;
    }
  }
  if (openMenu) await page.locator(entry.button).waitFor({ state: 'visible', timeout: 30000 });
  await page.waitForTimeout(450);
  const state = await snapshot();
  assert.equal(state.open, false, 'Discard restoration must enter ordinary mode, not resume Demo.');
  assert.equal(state.practice, false, 'Reload must not reconstruct practice.');
  assert.equal(state.nativeSurfaces, 0, 'No demo native scene survives reload.');
  return state;
}

          }
          await writeFile(join(evidenceDirectory, 'credential-' + index + '.html.txt'), row.credentialBytes, { flag: 'wx' });
          await writeFile(join(evidenceDirectory, 'served-' + index + '.html.txt'), row.servedBytes, { flag: 'wx' });
          index++;
        }
        report.observationFixture.artifacts = { directory: evidenceDirectory, originals: Object.fromEntries(originalPins), servedVariants: index };
        } catch (error) {
          report.complete = false;
          report.failure ??= browserErrorDescription(error);
          report.secondaryDiagnostics ??= [];
          report.secondaryDiagnostics.push({ stage: 'observation-artifact-write', error: browserErrorDescription(error) });
        }
      }
      // Bind failed attempts too, after browser and preview cleanup. A source
      // mismatch must fail qualification without replacing its primary failure.
      report.sourceUnchanged = false;
      try {
        const afterRuntime = await runtimeFingerprint();
        report.runtimeBinding.after = afterRuntime;
        assertRuntimeFingerprint(afterRuntime, 'After cleanup');
        assert.deepEqual(afterRuntime, beforeRuntime, 'Full canonical runtime remains unchanged.');
        assert.equal(report.runtimeBinding.beforeVerified, true, 'Runtime was admitted before launch.');
        report.afterHashes = await sourceHashes();
        assert.deepEqual(report.afterHashes, beforeHashes, 'Scoped source files remain unchanged.');
        assert.deepEqual(JSON.parse(await readFile(manifestPath)), manifest, 'Source declaration remains frozen.');
        report.headAfter = execFileSync('git', ['rev-parse', 'HEAD'], { encoding: 'utf8' }).trim();
        assert.equal(report.headAfter, args.head, 'Held HEAD remains unchanged.');
        assert.equal(sha(await readFile(new URL(import.meta.url))), report.probe.sha256, 'The probe remains unchanged.');
        const afterBuildInput = await fingerprintDemoSources(process.cwd(), ['scripts', 'authoring']);
        report.buildInputBinding.after = afterBuildInput;
        assert.equal(report.buildInputBinding.beforeVerified, true, 'Build inputs were bound before launch.');
        assert.deepEqual(afterBuildInput, beforeBuildInput, 'Complete canonical scripts/authoring build inputs remain unchanged.');
        if (args['emitted-preview']) {
          const afterEvidence = await emittedEvidenceSnapshot(process.cwd(), args['emitted-preview']);
          report.emittedEvidenceBinding.after = afterEvidence;
          assert.equal(report.emittedEvidenceBinding.beforeVerified, true, 'Emitted evidence was admitted before launch.');
          assert.deepEqual(afterEvidence, beforeEmittedEvidence, 'Exact emitted capture/audit/inventory bytes remain unchanged.');
          report.emittedEvidenceBinding.afterVerified = true;
        }
        report.buildInputBinding.afterVerified = true;
        report.runtimeBinding.afterVerified = true;
        report.sourceUnchanged = true;
      } catch (error) {
        report.complete = false;
        report.failure ??= browserErrorDescription(error);
        report.secondaryDiagnostics ??= [];
        if (report.secondaryDiagnostics.length < 8)
          report.secondaryDiagnostics.push({ stage: 'source-binding', error: browserErrorDescription(error) });
      }
      try {
        assertWithinDeadline();
        assert.ok(fixture, 'Observation fixture must have been admitted.');
        report.observationFixture = { ...report.observationFixture, ...fixture.summary() };
        report.strictDiagnosticGate = assertDiscardObservationHealth({ summary: report.observationFixture,
          errors: report.errors, errorOverflow: report.errorOverflow, events });
        report.finalObservationGate = { checkedAfterBrowserAndServerCleanup: true, passed: true };
      } catch (error) {
        report.complete = false;
        report.failure ??= browserErrorDescription(error);
        report.finalObservationGate = { checkedAfterBrowserAndServerCleanup: true, passed: false,
          error: browserErrorDescription(error) };
      }
      await requireFreshReport();
      await writeQualificationReport(...writeArgs);
    },
  });
  if (!result.complete) process.exitCode = 1;
}
