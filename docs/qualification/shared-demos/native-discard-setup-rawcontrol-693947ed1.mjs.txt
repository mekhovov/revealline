const bounded = async (fn, ms) => {
  let timer;
  try {
    return await Promise.race([Promise.resolve().then(fn), new Promise((_, reject) => {
      timer = setTimeout(() => reject(new Error(`Operation exceeded ${ms}ms.`)), ms);
    })]);
  } finally { clearTimeout(timer); }

// Raw transport exists only during a detached-target native discard. No auto-attach.
async function connectRaw(endpoint) {
  const socket = new WebSocket(endpoint);
  const pending = new Map();
  let serial = 0;
  const opened = new Promise((resolve, reject) => {
    socket.addEventListener('open', resolve, { once: true });
    socket.addEventListener('error', () => reject(new Error('Owned raw CDP WebSocket failed.')), { once: true });
  });
  const closed = new Promise(resolve => socket.addEventListener('close', () => {
    for (const { reject, timer } of pending.values()) {
      clearTimeout(timer);
      reject(new Error('Owned raw CDP connection closed.'));
    }
    pending.clear();
    resolve();
  }, { once: true }));
  socket.addEventListener('message', event => {
    const value = JSON.parse(event.data);
    if (value.id) {
      const row = pending.get(value.id);
      if (!row) return;
      pending.delete(value.id);
      clearTimeout(row.timer);
      if (value.error) row.reject(new Error(JSON.stringify(value.error)));
      else row.resolve(value.result);
    } else if (value.method === 'Runtime.exceptionThrown') {
      remember(report.errors, { stage, surface: 'owned-raw-control', native: value.params }, 16);
    }
  });
  try { await bounded(() => opened, 5000); }
  catch (error) { socket.close(); throw error; }
  return {
    send(method, params = {}, sessionId) {
      const id = ++serial;
      return new Promise((resolve, reject) => {
        const timer = setTimeout(() => {
          pending.delete(id);
          reject(new Error(`Raw CDP ${method} exceeded 3000ms.`));
        }, 3000);
        pending.set(id, { resolve, reject, timer });
        socket.send(JSON.stringify({ id, method, params, ...(sessionId ? { sessionId } : {}) }));
      });
    },
    async close() {
      if (socket.readyState !== WebSocket.CLOSED) socket.close();
      await bounded(() => closed, 2000);
    },
  };
}
function readNativeDiscardRow(targetURL, scroll = false) {
  const roots = [document];
  for (let i = 0; i < roots.length; i++) {
    for (const element of roots[i].querySelectorAll('*')) {
      if (element.shadowRoot) roots.push(element.shadowRoot);
    }
    if (roots.length > 100) throw new Error('Unexpected native UI shadow-tree size.');
  }
  const tables = roots.flatMap(root => [...root.querySelectorAll('#tab-discards-info-table-body')]);
  if (tables.length !== 1) return { tables: tables.length, matches: 0 };
  const matches = [...tables[0].querySelectorAll('tr')].filter(row =>
    [...row.querySelectorAll('td')].some(cell => cell.textContent.trim() === targetURL));
  if (matches.length !== 1) return { tables: 1, matches: matches.length };
  const row = matches[0];
  const actions = [...row.querySelectorAll('[data-id]')].filter(node => node.textContent.trim() === '[Urgent Discard]');
  if (actions.length !== 1) return { tables: 1, matches: 1, actions: actions.length };
  const urgent = actions[0];
  if (scroll) urgent.scrollIntoView({ block: 'center', inline: 'center' });
  const rect = urgent.getBoundingClientRect();
  return { tables: 1, matches: 1, actions: 1,
    cells: [...row.querySelectorAll('td')].map(cell => cell.textContent),
    disabled: urgent.hasAttribute('disabled'), tabId: urgent.dataset.id,
    rect: { x: rect.x, y: rect.y, width: rect.width, height: rect.height },
    viewport: { width: window.innerWidth, height: window.innerHeight } };
}
