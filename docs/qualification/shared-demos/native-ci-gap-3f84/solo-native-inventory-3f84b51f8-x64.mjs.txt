import assert from 'node:assert/strict';
import { readFile, writeFile } from 'node:fs/promises';
import { execFileSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { inventorySources, runSoloCandidateWorker } from './linux-capture-draft/scripts/capture-solo-native-candidates.mjs';
import { fingerprintDemoSources } from '../scripts/demo-source-fingerprint.mjs';

const args = Object.fromEntries(process.argv.slice(2).map(value => value.replace(/^--/, '').split('=')));
assert.equal(args.run, 'approved');
assert.equal(process.version, 'v22.13.1');
assert.equal(process.arch, 'x64');
assert.equal(process.platform, 'darwin');
const sha = body => createHash('sha256').update(body).digest('hex');
const head = () => execFileSync('/usr/bin/arch', ['-arm64', '/usr/bin/git', '-c', 'maintenance.auto=false', '-c', 'gc.auto=0', 'rev-parse', 'HEAD'], {encoding:'utf8',timeout:15000}).trim();
const pin = async name => {const body=await readFile(name); return {path:name,sha256:sha(body),bytes:body.length};};
const collector='.cache/linux-capture-draft/scripts/capture-solo-native-candidates.mjs';
const original=await readFile(new URL(import.meta.url));
const report={format:'SharedDemoSoloNativeInventoryObservationV1',complete:false,started:new Date().toISOString(),
  environment:{version:process.version,platform:process.platform,arch:process.arch,versions:process.versions,executable:await pin(process.execPath)},
  headBefore:head(),sourceBefore:await fingerprintDemoSources(process.cwd()),buildInputsBefore:await fingerprintDemoSources(process.cwd(),['scripts','authoring']),
  observerSHA256:sha(original),collectorBefore:await pin(collector),sources:[],errors:[],
  scope:'Read-only catalog-derived full Solo strict native availability scan. Existing reviewed compatibility is separately observed but cannot satisfy strict availability. No recordings or catalog changes. Darwinx64 reproduction is not Linux qualification.'};
assert.equal(report.headBefore,args.head);assert.equal(report.sourceBefore.sha256,args.fingerprint);
const controller=new AbortController(),started=performance.now();
const interrupt=()=>{controller.abort(Error('Scan interrupted'));report.complete=false;process.exitCode=1;};
process.once('SIGINT',interrupt);process.once('SIGTERM',interrupt);
const timer=setTimeout(()=>controller.abort(Error('Full native scan15-minute deadline exceeded')),900000);
const persist=async()=>{const body=JSON.stringify(report,null,2)+'\n';assert(Buffer.byteLength(body)<=8*1024**2);await writeFile(args.report,body);};
try{
  const inventory=await inventorySources(process.cwd());
  report.requiredEntries=inventory.requiredEntries;report.requiredSources=inventory.sources.length;report.catalogs=inventory.catalogs;report.missingEntries=inventory.missing;
  report.sourceOrder=inventory.sources.map(({clip})=>clip.id);await persist();
  console.log(JSON.stringify({requiredEntries:report.requiredEntries.length,requiredSources:report.requiredSources,missingEntries:report.missingEntries}));
  for(const item of inventory.sources){
    controller.signal.throwIfAborted();
    const remaining=900000-(performance.now()-started);assert(remaining>0);
    report.activeSource=item.clip.id;await persist();
    let row;
    try{row=await runSoloCandidateWorker({root:process.cwd(),clip:item.clip,entry:item.entry,capture:false},{timeoutMs:Math.min(60000,remaining),signal:controller.signal});}
    catch(error){if(error.code==='worker-cleanup-failed')throw error;row={id:item.clip.id,strictUsable:false,failure:String(error.stack??error).slice(0,4096)};}
    assert.equal(row.id,item.clip.id);assert(!row.candidate);row.requiredEntry=item.requiredEntry;report.sources.push(row);report.activeSource=null;
    await persist();console.log(JSON.stringify({finished:report.sources.length,required:report.requiredSources,id:row.id,strict:row.strictUsable,reviewedCompatibility:row.reviewedCompatibilityUsable??false,failure:row.failure??null}));
  }
  report.unusableSources=report.sources.filter(row=>!row.strictUsable).map(row=>row.id);
  report.runtimeUnusableSources=report.sources.filter(row=>!row.strictUsable&&!row.reviewedCompatibilityUsable).map(row=>row.id);
  report.unusableEntries=report.requiredEntries.filter(({id})=>!report.sources.some(row=>row.id===id&&row.strictUsable)).map(({id})=>id);
  report.counts={requiredEntries:report.requiredEntries.length,sources:report.sources.length,strictUsable:report.sources.filter(row=>row.strictUsable).length,
    strictFailed:report.unusableSources.length,reviewedCompatibilityOnly:report.sources.filter(row=>!row.strictUsable&&row.reviewedCompatibilityUsable).length};
  report.complete=report.sources.length===report.requiredSources&&!report.sources.some(row=>row.failure)&&!controller.signal.aborted;
  report.strictCoveragePassed=report.complete&&!report.missingEntries.length&&!report.unusableSources.length;
}catch(error){report.complete=false;report.failure=String(error.stack??error).slice(0,8192);}
finally{
  clearTimeout(timer);
  try{report.headAfter=head();report.sourceAfter=await fingerprintDemoSources(process.cwd());report.buildInputsAfter=await fingerprintDemoSources(process.cwd(),['scripts','authoring']);
    report.observerSHA256After=sha(await readFile(new URL(import.meta.url)));report.collectorAfter=await pin(collector);
    assert.equal(report.headAfter,report.headBefore);assert.deepEqual(report.sourceBefore,report.sourceAfter);assert.deepEqual(report.buildInputsBefore,report.buildInputsAfter);
    assert.equal(report.observerSHA256,report.observerSHA256After);assert.deepEqual(report.collectorBefore,report.collectorAfter);report.sourceUnchanged=true;
  }catch(error){report.complete=false;report.sourceUnchanged=false;report.bindingFailure=String(error.stack??error).slice(0,4096);}
  report.elapsedMs=performance.now()-started;if(report.elapsedMs>=900000||controller.signal.aborted)report.complete=false;
  report.unscannedSources=(report.sourceOrder??[]).filter(id=>!report.sources.some(row=>row.id===id));
  report.finished=new Date().toISOString();await writeFile(args.report.replace(/\.json$/,'.mjs.txt'),original,{flag:'wx'});await persist();
  process.removeListener('SIGINT',interrupt);process.removeListener('SIGTERM',interrupt);
  if(!report.complete)process.exitCode=1;
}
