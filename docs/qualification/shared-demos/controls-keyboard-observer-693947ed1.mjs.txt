async function fingerprint(directories = ['game', 'optional-practice', 'publishing']) {
  const paths = execFileSync('rg', ['--files', ...directories], { encoding: 'utf8', maxBuffer: 8 * 1024 * 1024 }).trim().split('\n')
    .filter(p => /\.(mjs|js|json|css|html)$/.test(p) && !p.includes('/test/') && !p.endsWith('.test.mjs') && !p.endsWith('/coverage.json')).sort();
  const hash = createHash('sha256'); let bytes = 0;
  for (const p of paths) { const body = await readFile(p); bytes += body.length; hash.update(p + '\0').update(Buffer.from(sha(body), 'hex')); }
  return { sha256: hash.digest('hex'), files: paths.length, bytes };
}
