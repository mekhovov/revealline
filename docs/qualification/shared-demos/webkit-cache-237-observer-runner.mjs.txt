function observe(raw, direction, transport, receivedAt) {
  if (raw.method === 'Target.dispatchMessageFromTarget' || raw.method === 'Target.sendMessageToTarget') {
    try { return observe(JSON.parse(raw.params.message), direction, { ...transport, pageProxyId: raw.pageProxyId, targetId: raw.params.targetId }, receivedAt); }
    catch { report.oversizedOrUnparseable++; return; }

    const key = JSON.stringify([transport.targetId, raw.id]);
    sent.set(key, { method: raw.method, id: raw.id, contextId: raw.params?.contextId ?? null,
      expression: String(raw.params?.expression ?? raw.params?.functionDeclaration ?? '').slice(0, 180) });
    if (sent.size > 300) sent.delete(sent.keys().next().value);
  }

  const isFailedEvaluation = request && (raw.error || raw.result?.wasThrown || raw.result?.exceptionDetails);
  if (!selected.has(raw.method) && !isFailedEvaluation) return;

    const message = raw.params?.message;
    if (message?.text?.startsWith('CACHE_OWNER_TRACE:') || message?.text?.includes('__pw_binding_call__')) return;
  }
  if (report.events.length >= 500) { report.overflow++; return; }
  report.events.push({ receivedAt, direction, ...transport, ...raw, ...(isFailedEvaluation ? { associatedRequest: request } : {}) });
}
