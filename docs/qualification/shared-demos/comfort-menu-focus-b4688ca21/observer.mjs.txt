async function fingerprint(directories = ['game', 'optional-practice', 'publishing']) {
  const paths = execFileSync('rg', ['--files', ...directories], { encoding: 'utf8', maxBuffer: 8 * 1024 * 1024 }).trim().split('\n')
    .filter(p => /\.(mjs|js|json|css|html)$/.test(p) && !p.includes('/test/') && !p.endsWith('.test.mjs') && !p.endsWith('/coverage.json')).sort();
  const hash = createHash('sha256'); let bytes = 0;
  for (const p of paths) { const body = await readFile(p); bytes += body.length; hash.update(p + '\0').update(Buffer.from(sha(body), 'hex')); }
  return { sha256: hash.digest('hex'), files: paths.length, bytes };
}

function decodedProfile(value) {

  assert.equal(decoded.warning, '', 'Profile must decode without repair/fallback.');
  assert.equal(decoded.recovery, null);

function assertComfortPersistence(before, after, display, menu) {
  const allowed = new Set([DISPLAY_PREFERENCES_KEY, MENU_ANIMATION_KEY, report.seededProfileKey]);
  const withoutAllowed = value => Object.fromEntries(Object.entries(value).filter(([key]) => !allowed.has(key)));
  assert.deepEqual(after.sessionStorage, before.sessionStorage, 'Explicit comfort actions never alter session persistence.');
  assert.deepEqual(withoutAllowed(after.localStorage), withoutAllowed(before.localStorage), 'Explicit comfort actions change only the shared display/menu stores and legacy profile mirror.');
  assert.deepEqual(JSON.parse(after.localStorage[DISPLAY_PREFERENCES_KEY]), display);
  assert.equal(after.localStorage[MENU_ANIMATION_KEY], menu);
  const previous = decodedProfile(before), current = decodedProfile(after);
  assert.equal(current.generation, previous.generation, 'Comfort does not replace the player profile generation.');
  assert.deepEqual(current.library, { ...previous.library, preferences: { ...previous.library.preferences, ...display } }, 'Only the three display preferences may change in the ordinary profile.');
}
