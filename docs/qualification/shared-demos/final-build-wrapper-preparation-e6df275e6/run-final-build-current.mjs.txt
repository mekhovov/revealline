async function fingerprint(directories) {
  const paths = execFileSync('rg', ['--files', ...directories], { encoding: 'utf8', maxBuffer: 8 * 1024 * 1024 }).trim().split('\n').filter(sourceFilter).sort();
  const hash = createHash('sha256'); let bytes = 0;
  for (const path of paths) { const body = await readFile(path); bytes += body.length; hash.update(path + '\0').update(createHash('sha256').update(body).digest()); }
  return { sha256: hash.digest('hex'), files: paths.length, bytes };
}
