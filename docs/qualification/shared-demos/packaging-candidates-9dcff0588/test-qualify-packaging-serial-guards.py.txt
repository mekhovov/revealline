selected=[x for x in main.body if isinstance(x,ast.FunctionDef) and x.name in {'interrupt','run_job'}]
assert len(selected)==2

def require(v,m):

counts=0
for signum in (2,15):
 events=[]; report={'jobs':[]}
 scope={'signal':Signal(),'report':report,'deferred_signal':{'active':False,'pending':None},'stamp':lambda:'now',
        'root':Path('/source'),'output':Path('/external/owned'),'node':Path('/node'),'env':{},'before':{'sourceRevision':'pin'},

        'args':SimpleNamespace(company_child_seconds=30,optional_child_seconds=30,base_path='/revealline/'),

 def popen(*a,**kw):
  events.append('spawned');scope['interrupt'](signum,None);events.append('spawn-returned');return SimpleNamespace(stdout=None,poll=lambda:0)

 try:scope['run_job']('company','fixture',time.monotonic()+30)
 except RuntimeError as e:assert 'did not finish cleanly' in str(e)
 else:raise AssertionError('Interrupted job must fail')
 assert events==['spawned','spawn-returned','owner-attached','owner-captured','log-start','cleanup'],events
