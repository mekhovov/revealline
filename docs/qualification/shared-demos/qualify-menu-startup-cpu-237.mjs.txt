function summarizeCPU(profile) {
  const nodes = new Map(profile.nodes.map((node) => [node.id, node]));
  const parents = new Map();
  for (const node of profile.nodes) for (const child of node.children ?? []) parents.set(child, node.id);
  const self = new Map(), inclusive = new Map(), stacks = new Map();
  const frame = (id) => { const n = nodes.get(id); return n ? { id, ...n.callFrame } : { id }; };
  for (let i = 0; i < (profile.samples ?? []).length; i++) {
    const id = profile.samples[i], micros = profile.timeDeltas?.[i] ?? 0;
    self.set(id, (self.get(id) ?? 0) + micros);
    const stack = []; let at = id;
    while (at !== undefined && stack.length < 80) {
      stack.push(at); inclusive.set(at, (inclusive.get(at) ?? 0) + micros); at = parents.get(at);
    }
    const key = stack.join(','); const prior = stacks.get(key);
    if (prior) prior.micros += micros; else stacks.set(key, { ids: stack, micros });
  }
  const top = (map) => [...map].sort((a,b) => b[1]-a[1]).slice(0,40).map(([id,micros]) => ({ micros, frame: frame(id) }));
  return { startTime: profile.startTime, endTime: profile.endTime, durationMicros: profile.endTime-profile.startTime,
    nodes: profile.nodes.length, samples: profile.samples?.length ?? 0, samplingIntervalMicros:1000,
    topSelf: top(self), topInclusive: top(inclusive),
    topStacks: [...stacks.values()].sort((a,b) => b.micros-a.micros).slice(0,20).map(({ids,micros}) => ({ micros, frames: ids.map(frame) })) };
}
async function finishCPU() {
  if (!cpuStarted || cpuFinished) return;
  cpuFinished = true;
  try {
    const { profile } = await cpuSession.send('Profiler.stop');
    report.cpuStopped = new Date().toISOString();
    report.cpu = summarizeCPU(profile);
    const bytes = JSON.stringify(profile); report.cpu.rawProfileBytes = Buffer.byteLength(bytes);
    if (Buffer.byteLength(bytes) <= 2097152) {
      const path = reportPath.replace(/\.json$/, '-whole-cpu.json');
      await writeFile(path, bytes, { flag: 'wx' });
      report.cpu.rawProfile = { path, sha256: sha(bytes) };
    } else report.cpu.rawProfileOmitted = 'Exceeds2MiB bound; bounded aggregate retained.';
    const offset = report.cpuClock.timestampMicros - (report.cpuClock.wallBefore + report.cpuClock.wallAfter) * 500;
    const times = []; let at = profile.startTime;
    for (const delta of profile.timeDeltas ?? []) { at += delta; times.push(at); }
    const intervals = [];
    for (const attempt of report.menuDiagnostics) {
      const windows = [{ name: 'menu-attempt', start: Date.parse(attempt.started), end: Date.parse(attempt.finished) }];
      for (const [index, task] of (attempt.observation?.longTasks ?? []).entries()) {
        const origin = attempt.before.wall - attempt.before.at;
        windows.push({ name: `observed-long-task-${index}`, start: origin + task.startTime, end: origin + task.startTime + task.duration });
      }
      for (const window of windows) {
        const start = window.start*1000+offset, end = window.end*1000+offset;
        const indexes = times.map((time,index) => time >= start && time <= end ? index : -1).filter((index) => index >= 0);
        const samples = indexes.map((index) => profile.samples[index]);
        const subset = { nodes: profile.nodes, samples, timeDeltas: indexes.map((index) => profile.timeDeltas[index]), startTime: start, endTime: end };
        intervals.push({ stage: attempt.stage, name: window.name, wallStart: window.start, wallEnd: window.end,
          clockUncertaintyMicros: report.cpuClock.uncertaintyMicros, ...summarizeCPU(subset) });
      }
    }
    report.cpuIntervals = intervals;
  } catch (error) { report.cpuError = browserErrorDescription(error); }
}

async function ordinaryEntry(entry, { openMenu = false } = {}) {
  const password = page.getByLabel('Password', { exact: true });
  if (await password.isVisible()) {
    await password.fill('fixture-password');
    await page.getByRole('button', { name: 'Unlock', exact: true }).click();
  }
  const menu = {
    solo: '#shell-menu',
    snake: '#snake-action-menu',
    sim: '#worlds-shell-action-menu',
    survival: '#overflight-action-menu',
  }[entry.mode];
  await page.waitForFunction(
    ({ button, menu }) => {
      if (globalThis.RevealLineBoot && document.documentElement.dataset.bootState !== 'ready')
        return false;
      const visible = (node) =>
        !!node?.getClientRects().length && getComputedStyle(node).visibility !== 'hidden';
      return (
        visible(document.querySelector(button)) ||
        (visible(document.querySelector(menu)) && !document.querySelector('dialog[open]'))
      );
    },
    { button: entry.button, menu },
    { timeout: 30000 },
  );
  if (openMenu && !(await page.locator(entry.button).isVisible())) {
    const menuDiagnostic = { mode: entry.mode, stage, started: new Date().toISOString() };
    report.menuDiagnostics.push(menuDiagnostic);
    menuDiagnostic.before = await page.evaluate(() => window.demoMenuDiagnostic.start());
    try {
      await page.locator(menu).click({ timeout: 3000 });
      menuDiagnostic.clickResolved = true;
    } catch (error) {
      menuDiagnostic.clickError = browserErrorDescription(error);
      if (!(await page.locator(entry.button).isVisible())) throw error;
    } finally {
      menuDiagnostic.finished = new Date().toISOString();
      try { menuDiagnostic.observation = await page.evaluate(() => window.demoMenuDiagnostic.stop()); }
      catch (error) { menuDiagnostic.observerError = browserErrorDescription(error); }
      if (menuDiagnostic.clickError) {
        try { await page.screenshot({ path: reportPath.replace(/\.json$/, '-menu.png'), timeout: 2000 }); }
        catch (error) { menuDiagnostic.screenshotError = browserErrorDescription(error); }
      }
    }
  }
  if (openMenu) await page.locator(entry.button).waitFor({ state: 'visible', timeout: 30000 });
  await page.waitForTimeout(450);
  const state = await snapshot();
  assert.equal(state.open, false, 'Reload must enter ordinary mode, not resume Demo.');
  assert.equal(state.practice, false, 'Reload must not reconstruct practice.');
  assert.equal(state.nativeSurfaces, 0, 'No demo native scene survives reload.');
  return state;
}

          await new Promise((resolve, reject) => {
            const closeTimer = setTimeout(
              () => reject(new Error('Preview cleanup exceeded 2000ms.')),
              2000,
            );
            preview.server.close((error) => {
              clearTimeout(closeTimer);
              if (error) reject(error);
              else resolve();
            });
          });
          report.previewClosed = true;
        } catch (error) {
          report.previewClosed = false;
          report.complete = false;
          report.failure ??= browserErrorDescription(error);
        }
      }
      await writeQualificationReport(...writeArgs);
    },
  });
  if (!result.complete) process.exitCode = 1;
}
