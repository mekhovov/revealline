import {execFileSync} from 'node:child_process';
import {fileURLToPath} from 'node:url';
const root=fileURLToPath(new URL('../../',import.meta.url));
const source='41d0b908e4c7c5f1cfbbd24994f2e96381332b28';
const cache=new Map();
export function sourceFor(name){
 if(!name||name.startsWith('/')||name.split('/').some(x=>!x||x==='..'))throw Error('Invalid source path');
 if(!cache.has(name))cache.set(name,execFileSync('git',['--no-lazy-fetch','--no-optional-locks','show',source+':'+name],{cwd:root,maxBuffer:32*1024*1024,env:{...process.env,GIT_NO_LAZY_FETCH:'1',GIT_OPTIONAL_LOCKS:'0',GIT_TERMINAL_PROMPT:'0'}}));
 return Buffer.from(cache.get(name));
}
