export * from 'node:fs/promises';
import {readFile as actualReadFile} from 'node:fs/promises';
import {fileURLToPath,URL as NativeURL} from 'node:url';
import {relative,resolve} from 'node:path';
import {sourceFor} from './patch-source.mjs';
const candidate=fileURLToPath(new URL('./candidate/',import.meta.url));
export async function readFile(file,options){
 const path=file instanceof NativeURL?fileURLToPath(file):resolve(String(file)),name=relative(candidate,path);
 if(name.startsWith('..'))return actualReadFile(file,options);
 const bytes=sourceFor(name),encoding=typeof options==='string'?options:options?.encoding;
 return encoding?bytes.toString(encoding):bytes;
}

import * as actual from 'node:fs/promises';


const entries=JSON.parse(await actual.readFile(new URL('./tree-index.json',import.meta.url),'utf8'));
const dirs=new Map([['',new Set()]]);
for(const name of Object.keys(entries)){
 const parts=name.split('/');let parent='';
 for(let i=0;i<parts.length;i++){
  if(!dirs.has(parent))dirs.set(parent,new Set());dirs.get(parent).add(parts[i]);
  parent=parent?parent+'/'+parts[i]:parts[i];
 }
}
function virtualName(file){const p=file instanceof NativeURL?fileURLToPath(file):resolve(String(file));const name=relative(candidate,p);return name.startsWith('..')?null:name;}
export async function lstat(file,...args){
 const name=virtualName(file);if(name===null)return actual.lstat(file,...args);
 const entry=entries[name],directory=dirs.has(name);
 if(!entry&&!directory){const e=Error('Missing declared Git fixture: '+name);e.code='ENOENT';throw e;}
 return {get size(){return directory?0:sourceFor(name).length;},isDirectory:()=>directory,isFile:()=>entry?.kind==='blob'&&entry.mode!=='120000',isSymbolicLink:()=>entry?.mode==='120000'};
}
export async function stat(file,...args){return virtualName(file)===null?actual.stat(file,...args):lstat(file,...args);}
export async function readdir(file,...args){const name=virtualName(file);if(name===null)return actual.readdir(file,...args);if(args.length)throw Error('Unsupported Git fixture readdir options');if(!dirs.has(name))throw Error('Missing fixture directory');return [...dirs.get(name)];}
export default {...actual,readFile,lstat,stat,readdir};
